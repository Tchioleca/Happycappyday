  <!-- Rive runtime (unpkg CDN) -->
  <script src="https://unpkg.com/rive-js@latest/dist/rive.min.js"></script>
  <script>
    (function () {
      const canvas = document.getElementById('riveCanvas');

      // === EDIT THESE to match your Rive file exactly ===
      const riveFile = 'happycappyday_(1).riv';      // exact filename in repo
      const stateMachineName = 'cappystate';        // exact state machine name
      const triggerName = 'click';                  // exact trigger input name
      // ==================================================

      const riveInstance = new rive.Rive({
        src: riveFile,
        canvas: canvas,
        autoplay: true,
        stateMachines: stateMachineName,
        onLoad: () => {
          try {
            const inputs = riveInstance.stateMachineInputs(stateMachineName) || [];
            console.log('Rive inputs:', inputs.map(i => i.name));

            const triggerInput = inputs.find(i => i.name === triggerName);
            if (!triggerInput) {
              console.warn('Trigger not found. Inputs list:', inputs.map(i => i.name));
              return;
            }

            canvas.addEventListener('pointerdown', () => {
              if (typeof triggerInput.fire === 'function') {
                triggerInput.fire(); // ideal for trigger inputs
              } else if ('value' in triggerInput) {
                // fallback (boolean/number style inputs)
                triggerInput.value = true;
                setTimeout(() => triggerInput.value = false, 80);
              }
            });

            console.log('Rive loaded. Trigger hooked:', triggerName);
          } catch (err) {
            console.error('Error hooking inputs:', err);
          }
        }
      });
    })();
  </script>
